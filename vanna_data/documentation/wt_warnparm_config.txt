# Table: public.wt_warnparm_config — 监测预警参数配置表

# Column documentation:
wt_warnparm_config.id: id
wt_warnparm_config.indicator_code: 监测指标编号
wt_warnparm_config.l1_value: 一级
wt_warnparm_config.l2_value: 二级
wt_warnparm_config.l3_value: 三级
wt_warnparm_config.l4_value: 四级
wt_warnparm_config.l5_value: 五级
wt_warnparm_config.zero: 零点核查
wt_warnparm_config.zero_drift: 24小时零点漂移
wt_warnparm_config.span: 跨度核查
wt_warnparm_config.span_drift: 24小时跨度漂移
wt_warnparm_config.compare: 比较方式：0 无；1 ≤；2 ≥
wt_warnparm_config.type: 0 河流，1 湖泊
wt_warnparm_config.possible_value: 可能发生污染的告警值
wt_warnparm_config.remove_warn_value: 消除预警的参数值
wt_warnparm_config.frequency: 监测频率（小时计）
wt_warnparm_config.create_by: 创建人
wt_warnparm_config.create_time: 创建时间
wt_warnparm_config.update_by: 更新人
wt_warnparm_config.update_time: 更新时间
wt_warnparm_config.del_flag: 删除标记：0 未删除、1 已删除
