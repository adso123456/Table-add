# Table: public.rs_outlet_info_v2 — 排污口管理

rs_outlet_info_v2.id: 主键ID
rs_outlet_info_v2.outlet_code_national: 排污口编码(国家统一赋码)
rs_outlet_info_v2.outlet_code_local: 排污口编码(省市系统自有编码)
rs_outlet_info_v2.is_key_river_outlet: 是否为重点河湖排污口
rs_outlet_info_v2.key_river_name: 重点河湖名称
rs_outlet_info_v2.river_system: 所属水系
rs_outlet_info_v2.province_city_district: 省市区县
rs_outlet_info_v2.township: 乡镇(街道)
rs_outlet_info_v2.outlet_name: 排污口名称
rs_outlet_info_v2.watershed_name: 流域名称
rs_outlet_info_v2.outlet_category: 排污口分类
rs_outlet_info_v2.is_lake_inflow: 是否为入湖河流
rs_outlet_info_v2.is_water_func_zone: 是否位于水功能区
rs_outlet_info_v2.level1_water_func_zone: 一级水功能区名称
rs_outlet_info_v2.level1_water_quality_target: 一级水功能区水质目标
rs_outlet_info_v2.level2_water_func_zone: 二级水功能区名称
rs_outlet_info_v2.level2_water_quality_target: 二级水功能区水质目标
rs_outlet_info_v2.is_water_env_func_zone: 是否位于水环境功能区
rs_outlet_info_v2.water_env_func_zone_name: 水环境功能区名称
rs_outlet_info_v2.water_env_func_zone_target: 水环境功能区目标
rs_outlet_info_v2.national_section_name: 国控断面名称
rs_outlet_info_v2.provincial_section_name: 省控断面名称
rs_outlet_info_v2.municipal_section_name: 市控断面名称
rs_outlet_info_v2.river_chief_info: 河长信息
rs_outlet_info_v2.create_by: 创建人
rs_outlet_info_v2.create_time: 创建时间
rs_outlet_info_v2.update_by: 修改人
rs_outlet_info_v2.update_time: 修改时间
rs_outlet_info_v2.del_flag: 删除标志：0-正常，1-已删除
rs_outlet_info_v2.geom: 图形信息
