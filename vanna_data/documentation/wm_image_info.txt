# Table: public.wm_image_info — 档案图片信息表：用于保存各种站点、断面、污染源的图片信息

# Column documentation:
wm_image_info.id: 主键ID
wm_image_info.pid: 照片关联的id
wm_image_info.image_url: 图片地址
wm_image_info.image_formats: 图片格式
wm_image_info.image_type: 图片类型：0-未知、1-断面、2-自动站......
wm_image_info.create_by: 创建人
wm_image_info.create_time: 创建时间
wm_image_info.update_by: 更新人
wm_image_info.update_time: 更新时间
wm_image_info.del_flag: 删除标记: 0-未删除、1-已删除
wm_image_info.image_name: 文件名称：原始文件名
wm_image_info.image_tag: 图片标签
